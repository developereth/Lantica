// Cloudflare Worker - Lantica 1960 Menu API // GitHub token stored as Environment Variable - NEVER exposed export default { async fetch(request, env) { const url = new URL(request.url); const path = url.pathname; } const adminKey = request.headers.get('X-Admin-Key'); const isAdmin = adminKey === env.ADMIN_PASSWORD; // GET /api/menu - Read menu if (path === '/api/menu' && request.method === 'GET') { try { const githubUrl = `https://api.github.com/repos/${env.GITHUB_REPO}/contents/menu.json`; const response = await fetch(githubUrl, { headers: { 'Authorization': `Bearer ${env.GITHUB_TOKEN}` } }); if (!response.ok) throw new Error('GitHub API error'); const data = await response.json(); const content = JSON.parse(atob(data.content)); return new Response(JSON.stringify(content), { headers: { 'Content-Type': 'application/json', ...corsHeaders } }); } catch (error) { return new Response(JSON.stringify({ error: error.message }), { status: 500, headers: corsHeaders }); } } // POST /api/menu - Update menu (requires admin password) if (path === '/api/menu' && request.method === 'POST') { if (!isAdmin) { return new Response(JSON.stringify({ error: 'Unauthorized' }), { status: 401, headers: corsHeaders }); const getResponse = await fetch(getUrl, { headers: { 'Authorization': `Bearer ${env.GITHUB_TOKEN}` } }); let sha = null; if (getResponse.ok) { const data = await getResponse.json(); sha = data.sha; } const putResponse = await fetch(getUrl, { method: 'PUT', headers: { 'Authorization': `Bearer ${env.GITHUB_TOKEN}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ message: 'Menu update via Cloudflare Worker', content: content, sha: sha, branch: 'main' }) }); if (!putResponse.ok) throw new Error('Failed to save'); return new Response(JSON.stringify({ success: true }), { headers: { 'Content-Type': 'application/json', ...corsHeaders } }); } catch (error) { return new Response(JSON.stringify({ error: error.message }), { status: 500, headers: corsHeaders });
